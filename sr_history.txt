    Member lookup status & realtime transcript UI
    
    Improve member-lookup UX and realtime transcription handling across backend and frontend. Key changes:
    
    - agent/prompts.py: Add guidance to stall naturally (ask claim questions) while a policy lookup is still pending.
    - main.py: Emit `member_lookup_status` websocket events when fast-path lookups (policy or phone) haven't yet found a member; overhaul language detection to combine long-term and short-term signals and always include English as a baseline.
    - frontend/public/pcm-processor.js: Reduce PCM hold window from 300ms to 200ms to tighten channel switching.
    - frontend/src/hooks/useSpeechRecognition.js: Tweak Deepgram parameters (endpointing → 1000, add utterance_end_ms=1200).
    - frontend/src/hooks/useWebSocket.js: Add memberLookupStatus state and addTranscript helper to merge consecutive finalized same-speaker segments and display Deepgram partials/finals instantly; handle `member_lookup_status` messages and clear status on `member_profile`.
    - frontend/src/pages/CallPage.jsx & frontend/src/components/MemberCard.jsx: Wire lookup status into the UI, display a searching indicator/message in the MemberCard, and use addTranscript to show Deepgram results immediately while still sending data to the backend for AI processing.
    
    Overall this improves perceived responsiveness during lookups, reduces transcript fragmentation, and makes STT language selection more robust.

+/**
+ * Custom hook for Azure Speech SDK with real-time transcription

+            const audioConfig = SpeechSDK.AudioConfig.fromDefaultMicrophoneInput();

+            );
+
+            transcriberRef.current = transcriber;
+        } catch (err) {
+            console.error('Azure Speech init error:', err);
+            setError(err.message);
+        }
+    }, [onTranscript]);
+
+    const stopListening = useCallback(() => {
+        if (transcriberRef.current) {
+            transcriberRef.current.stopTranscribingAsync(
+                () => {
+                    console.log('🎙️ Azure Speech: transcription stopped');
+                    transcriberRef.current?.close();
+                    transcriberRef.current = null;
+                },
+                (err) => console.error('Error stopping transcription:', err)
+            );
+        }
+        setIsListening(false);
+    }, []);
+
+    const toggleListening = useCallback(() => {
+        if (isListening) {
+            stopListening();
+        } else {
+            startListening();
+        }
+    }, [isListening, startListening, stopListening]);
+
+    return {
+        isListening,
+        error,
+        startListening,
+        stopListening,
+        toggleListening,
+    };
+}
